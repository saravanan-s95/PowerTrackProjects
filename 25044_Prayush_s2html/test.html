11 int main(int argc, char *argv[]) {
12     FILE *sfp, *dfp; // source and destination file descriptors
13     pevent_t *event;
14     char dest_file[100];
15     char *src_file = NULL;
16     char *out_file = NULL;
17     int line_num_enabled = 0;
18 
19     if (argc < 2) {
20         printf("\nError ! please enter file name and mode\n");
21         printf("Usage: %s <file name> [output file] [-n]\n", argv[0]);
22         printf("Example : ./s2html abc.c abc.html -n\n\n");
23         return 1;
24     }
25 
26     /* Parse command line arguments */
27     for (int i = 1; i < argc; i++) {
28         if (strcmp(argv[i], "-n") == 0) {
29             line_num_enabled = 1;
30         } else if (src_file == NULL) {
31             src_file = argv[i];
32         } else if (out_file == NULL) {
33             out_file = argv[i];
34         }
35     }
36 
37     if (src_file == NULL) {
38         printf("Error! No input file specified\n");
39         return 1;
40     }
41 
42 #ifdef DEBUG
43     printf("File to be opened : %s\n", src_file);
44 #endif
45 
46     /* open the file */
47     if (NULL == (sfp = fopen(src_file, "r"))) {
48         printf("Error! File %s could not be opened\n", src_file);
49         return 2;
50     }
51 
52     /* Check for output file */
53     if (out_file != NULL) {
54         if (strstr(out_file, ".html") || strstr(out_file, ".htm")) {
55             snprintf(dest_file, sizeof(dest_file), "%s", out_file);
56         } else {
57             snprintf(dest_file, sizeof(dest_file), "%s.html", out_file);
58         }
59     } else {
60         snprintf(dest_file, sizeof(dest_file), "%s.html", src_file);
61     }
62 
63     /* open dest file */
64     if (NULL == (dfp = fopen(dest_file, "w"))) {
65         printf("Error! could not create %s output file\n", dest_file);
66         fclose(sfp);
67         return 3;
68     }
69 
70     /* configure line numbers */
71     set_line_number_flag(line_num_enabled);
72 
73     /* write HTML starting Tags */
74     html_begin(dfp, HTML_OPEN);
75 
76     /* Read from src file convert into html and write to dest file */
77     do {
78         event = get_parser_event(sfp);
79         /* call source_to_html */
80         source_to_html(dfp, event);
81     } while (event->type != PEVENT_EOF);
82 
83     /* Call start_or_end_conv function for ending the conversion */
84     html_end(dfp, HTML_CLOSE);
85 
86     printf("\nOutput file %s generated\n", dest_file);
87     /* close file */
88     fclose(sfp);
89     fclose(dfp);
90 
91     return 0;
92 }
